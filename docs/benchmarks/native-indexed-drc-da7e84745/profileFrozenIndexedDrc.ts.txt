  traceClearance: RELAXED_DRC_OPTIONS.traceClearance ?? 0.1, viaClearance: RELAXED_DRC_OPTIONS.viaClearance ?? 0.1,

// These are two separate persistent same-source engines. Control work and all
// fixture decoding, assertions, source hashing and report serialization are
// outside the one sampled span. No native method or builtin is replaced.
const originalDescriptors = Object.getOwnPropertyDescriptors(Engine.prototype)
const oracle = new Engine(makeSrj(), makeOptions())
const expected = rawInputs.map(({ input }) => {
  const result = oracle.evaluate(structuredClone(input.routedTraces))
  return { result, stats: { ...oracle.lastRunStats } }
})
const profiledInputs = rawInputs.map(({ input }) => structuredClone(input.routedTraces))
const srj = makeSrj(), options = makeOptions()
const actual: { result: any; stats: any }[] = []
let engine: InstanceType<typeof Engine> | undefined
const report = profile((): void => {
  engine = new Engine(srj, options)
  for (const traces of profiledInputs) {
    const result = engine.evaluate(traces)
    actual.push({ result, stats: { ...engine.lastRunStats } })
  }
})
deepStrictEqual(Object.getOwnPropertyDescriptors(Engine.prototype), originalDescriptors)
deepStrictEqual(actual, expected)

type Frame = { name: string; sourceID: number }
type Trace = { frames: Frame[] }
const traces = report.stackTraces.traces as Trace[]
const urls = new Map(report.stackTraces.sources.map(source => [source.sourceID, source.url ?? ""]))
const rows = new Map<string, { url: string; name: string; selfSamples: number; inclusiveSamples: number }>()
const stacks = new Map<string, number>()
for (const trace of traces) {
  const seen = new Set<string>()
  for (let index = 0; index < trace.frames.length; index++) {
    const frame = trace.frames[index]!, url = urls.get(frame.sourceID) ?? "", key = `${url}:${frame.name}`
    let row = rows.get(key)
    if (!row) { row = { url, name: frame.name, selfSamples: 0, inclusiveSamples: 0 }; rows.set(key, row) }
    if (index === 0) row.selfSamples++
    if (!seen.has(key)) { row.inclusiveSamples++; seen.add(key) }
  }
  const stack = trace.frames.slice(0, 8).map(frame => `${urls.get(frame.sourceID) ?? ""}:${frame.name}`).join(" > ")
  stacks.set(stack, (stacks.get(stack) ?? 0) + 1)
}

}
